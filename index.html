1. KPI Card Color — Last Month
Returns Green, Red or Grey for conditional formatting. 
KPI Card Color Month =

VAR _CM =
    [Sales]

VAR _PM =
    CALCULATE(
        [Sales],
        DATEADD(
            'Date Table'[Date],
            -1,
            MONTH
        )
    )

VAR _perc =
    DIVIDE(
        _CM - _PM,
        _PM
    )

VAR _format =
    SWITCH(
        TRUE(),
        _perc > 0,
            "Green",
        _perc < 0,
            "Red",
        "Grey"
    )

RETURN
    _format

3. ScorePct — vs Last Year
Calculates ScorePct percentage change compared with the previous year. 
ScorePct LY =

VAR _CM =
    AVERAGE('powerbi_overall_vm_score'[ScorePct])

VAR _LY =
    CALCULATE(
        AVERAGE('powerbi_overall_vm_score'[ScorePct]),
        DATEADD(
            'powerbi_overall_vm_score'[ReportMonth],
            -1,
            YEAR
        )
    )

VAR _Change =
    DIVIDE(
        _CM - _LY,
        _LY
    )

RETURN
    SWITCH(
        TRUE(),
        _Change > 0,
            UNICHAR(11165) & " " & FORMAT(_Change, "0.0%"),
        _Change < 0,
            UNICHAR(11167) & " " & FORMAT(ABS(_Change), "0.0%"),
        "0.0%"
    )
